cv = StratifiedKFold(n_splits=N_SPLITS, shuffle=True, random_state=RANDOM_STATE)

oof_predictions = np.empty(len(y_train), dtype=object)

for fold, (fit_index, cv_index) in enumerate(cv.split(X_train, y_train), start=1):

    fold_pipeline = clone(base_pipeline)
    started = time.perf_counter()
    fold_pipeline.fit(X_fit, y_fit)
    fit_seconds = time.perf_counter() - started

    started = time.perf_counter()
    cv_prediction = fold_pipeline.predict(X_cv)
    predict_seconds = time.perf_counter() - started
    train_prediction = fold_pipeline.predict(X_fit)

    cv_scores = compute_metrics(y_cv, cv_prediction)
    train_scores = compute_metrics(y_fit, train_prediction)
    fitted_tree = fold_pipeline.named_steps['model']
    record = {
        'experiment_id': EXPERIMENT['id'],
        'fold': fold,

        'fit_seconds': fit_seconds,
        'predict_seconds': predict_seconds,

class NumericCorrelationFilter(BaseEstimator, TransformerMixin):
    # Drop redundant numeric columns using correlations learned at fit time.

    def __init__(self, threshold=0.98):
        self.threshold = threshold

    def fit(self, X, y=None):
        if not isinstance(X, pd.DataFrame):
            raise TypeError("NumericCorrelationFilter requires a pandas DataFrame.")

        self.feature_names_in_ = X.columns.to_numpy()
        self.numeric_columns_ = X.select_dtypes(include=["number", "bool"]).columns.tolist()
        self.drop_columns_ = []
        if len(self.numeric_columns_) > 1:
            corr = X[self.numeric_columns_].corr().abs()
            upper = corr.where(np.triu(np.ones(corr.shape), k=1).astype(bool))
            self.drop_columns_ = [
                column for column in upper.columns if (upper[column] > self.threshold).any()
            ]

        self.keep_columns_ = [
            column for column in X.columns if column not in self.drop_columns_
        ]
        return self

    def transform(self, X):
        if not hasattr(self, "keep_columns_"):
            raise RuntimeError("The filter must be fitted before transform.")
        return X.loc[:, self.keep_columns_].copy()

def signed_log(values):
    """
    Signed-log:
        sign(x) * log(1 + abs(x))

    Giữ dấu âm/dương và nén độ lớn.
    """

    values = np.asarray(
        values,
        dtype=np.float64
    )

    return (
        np.sign(values)
        * np.log1p(np.abs(values))
    )

def make_e12_pipeline():

    # Tương thích các phiên bản sklearn
    ohe_args = {
        "handle_unknown": "ignore"
    }

    if "sparse_output" in inspect.signature(
        OneHotEncoder
    ).parameters:
        ohe_args["sparse_output"] = False
    else:
        ohe_args["sparse"] = False

    # Signed-log đã được thực hiện trong notebook Processing.
    # StandardScaler được fit riêng theo từng fold CV.
    numeric_pipeline = Pipeline([
        ("scaler", StandardScaler())
    ])

    # Tiền xử lý toàn bộ đặc trưng
    preprocessor = ColumnTransformer(
        transformers=[
            (
                "numeric",
                numeric_pipeline,
                NUMERIC_FEATURES
            ),
            (
                "categorical",
                OneHotEncoder(**ohe_args),
                CATEGORICAL_FEATURES
            )
        ],
        remainder="drop",
        sparse_threshold=0
    )

    # MLP giữ nguyên Baseline
    mlp = MLPClassifier(
        hidden_layer_sizes=(64,),
        activation="relu",
        solver="adam",
        alpha=0.0001,
        learning_rate_init=0.001,
        batch_size=512,
        max_iter=60,
        n_iter_no_change=8,
        early_stopping=False,
        random_state=RANDOM_STATE,
        verbose=False
    )

    return Pipeline([
        ("preprocess", preprocessor),
        ("model", mlp)
    ])

# ========================================
# 3-FOLD CROSS-VALIDATION E12
# ========================================

cv = StratifiedKFold(
    n_splits=N_SPLITS,
    shuffle=True,
    random_state=RANDOM_STATE
)

fold_rows = []
per_class_rows = []
loss_curves = {}

for fold, (fit_idx, cv_idx) in enumerate(
    cv.split(X, y),
    start=1
):

    print("\n" + "=" * 55)
    print(f"E12 - FOLD {fold}/{N_SPLITS}")
    print("=" * 55)

    # Dữ liệu huấn luyện trong fold
    X_fit = X.iloc[fit_idx]
    y_fit = y.iloc[fit_idx]

    # Dữ liệu kiểm định trong fold
    X_cv = X.iloc[cv_idx]
    y_cv = y.iloc[cv_idx]

    # Khởi tạo Pipeline mới cho mỗi fold
    fold_pipeline = make_e12_pipeline()

    start_time = time.perf_counter()

    # Huấn luyện
    fold_pipeline.fit(X_fit, y_fit)

    elapsed = time.perf_counter() - start_time

    # Dự đoán
    pred_cv = fold_pipeline.predict(X_cv)
    pred_train = fold_pipeline.predict(X_fit)

    # Đánh giá
    cv_scores = evaluate(y_cv, pred_cv)
    train_scores = evaluate(y_fit, pred_train)

    trained_mlp = fold_pipeline.named_steps["model"]

    # Lưu Training Loss
    loss_curves[fold] = list(
        trained_mlp.loss_curve_
    )

    # Lưu kết quả từng fold
    fold_rows.append({
        "fold": fold,
        "n_train": len(X_fit),
        "n_cv": len(X_cv),
        "epochs": int(trained_mlp.n_iter_),
        "training_time_sec": float(elapsed),
        "final_train_loss": float(trained_mlp.loss_),

        **cv_scores,

        **{
            f"train_{m}": train_scores[m]
            for m in METRICS
        }
    })

    # Đánh giá theo từng lớp
    p, r, f, support = (
        precision_recall_fscore_support(
            y_cv,
            pred_cv,
            labels=LABELS,
            zero_division=0
        )
    )

    for label, precision, recall, f1_value, n in zip(
        LABELS,
        p,
        r,
        f,
        support
    ):

        per_class_rows.append({
            "fold": fold,
            "class": label,
            "precision": float(precision),
            "recall": float(recall),
            "f1": float(f1_value),
            "support": int(n)
        })

    print("Số mẫu Train:", len(X_fit))
    print("Số mẫu CV:", len(X_cv))
    print("Số epoch:", trained_mlp.n_iter_)
    print(f"Thời gian: {elapsed:.2f} giây")

    print("\n=== KẾT QUẢ FOLD ===")

    for metric in METRICS:
        print(
            f"{metric:18s}: {cv_scores[metric]:.6f}"
        )


# ========================================
# TỔNG HỢP 3 FOLD
# ========================================

cv_results = pd.DataFrame(fold_rows)
cv_per_class = pd.DataFrame(per_class_rows)

cv_summary = pd.DataFrame({
    "metric": METRICS,

    "mean": [
