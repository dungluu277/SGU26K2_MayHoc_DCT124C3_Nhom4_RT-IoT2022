def find_data_file(filename: str) -> Path:
    # Find the files independently of the notebook working directory.
    cwd = Path.cwd()
    candidates = [
        cwd / filename,
        cwd / "splitted data" / filename,
        cwd / ".." / "splitted data" / filename,
        cwd / ".." / ".." / "splitted data" / filename,
        cwd / "data" / filename,
    ]

    for parent in (cwd, *cwd.parents):
        candidates.append(parent / "splitted data" / filename)
        candidates.append(parent / "data" / "splitted data" / filename)

    for path in candidates:
        path = path.resolve()
        if path.is_file():
            return path

    searched = "\n".join(f"  - {path.resolve()}" for path in candidates)
    raise FileNotFoundError(
        f"Không tìm thấy {filename}. Các đường dẫn đã kiểm tra:\n{searched}"
    )

support = cm.sum(axis=1)
predicted_support = cm.sum(axis=0)
true_positive = np.diag(cm)
precision = np.divide(true_positive, predicted_support, out=np.zeros_like(true_positive, dtype=float), where=predicted_support != 0)
recall = np.divide(true_positive, support, out=np.zeros_like(true_positive, dtype=float), where=support != 0)
f1 = np.divide(2 * precision * recall, precision + recall, out=np.zeros_like(precision), where=(precision + recall) != 0)
per_class = pd.DataFrame({"precision": precision, "recall": recall, "f1_score": f1, "support": support}, index=labels)

Nhận xét chi tiết

Dữ liệu và quy trình. Dữ liệu dùng để đánh giá gồm 82,536 mẫu Train và 17,686 mẫu Validation, cùng 12 lớp. Tập Train có lớp DOS_SYN_Hping lớn nhất với 63,062 mẫu (76.41%), trong khi NMAP_FIN_SCAN ít nhất với 20 mẫu. Tập Validation có 13,513 mẫu DOS_SYN_Hping (76.41%) và chỉ 4 mẫu NMAP_FIN_SCAN. Tỷ lệ lớp lớn nhất/lớp nhỏ nhất lần lượt là 3,153.1:1 ở Train và 3,378.2:1 ở Validation. Cross-validation được thực hiện trên Train; Validation được giữ riêng để đánh giá sau cùng, Test không được dùng.

Thiết lập mô hình. Random Forest dùng 100 cây, random_state=42 và n_jobs=-1; các biến phân loại được one-hot encode. Ba-fold StratifiedKFold được chạy trên Train, sau đó pipeline được fit trên toàn bộ Train và dự đoán Validation.

Cross-validation. Accuracy trung bình=0.9979 ± 0.0001; macro-F1=0.9652 ± 0.0040; weighted-F1=0.9978; macro-precision=0.9772; macro-recall=0.9593. Độ lệch chuẩn macro-F1 nhỏ hơn các mô hình còn lại, cho thấy kết quả giữa ba fold ổn định hơn trong lần chạy này.

Validation độc lập. Accuracy=0.9984 (99.84%), macro-F1=0.9810, weighted-F1=0.9984, macro-precision=0.9973, macro-recall=0.9680. Ma trận nhầm lẫn có 17,657/17,686 dự đoán đúng và 29 dự đoán sai. Precision macro cao hơn recall macro, nghĩa là dự đoán dương tính nhìn chung chính xác nhưng một số mẫu thuộc các lớp vẫn bị bỏ sót.

Khác biệt giữa các lớp. Ba lớp F1 thấp nhất là NMAP_FIN_SCAN: F1=0.857, recall=0.750, support=4; Wipro_bulb: F1=0.952, recall=0.909, support=33; NMAP_UDP_SCAN: F1=0.990, recall=0.990, support=387. Lớp NMAP_FIN_SCAN chỉ có 4 mẫu trong Validation; recall 0.75 ở support 4 là bằng chứng rất hạn chế để khẳng định hiệu năng ổn định cho lớp này. Bảng confusion matrix và per-class metrics trong notebook/Excel cho thấy lỗi tập trung cụ thể ở đâu.

Kết luận. Trong ba baseline đã chạy trên cùng một phép chia dữ liệu, Random Forest đạt CV macro-F1 và Validation macro-F1 cao nhất. Notebook này không tính điểm trên tập Train sau fit, vì vậy không thể định lượng train-CV gap hoặc kết luận chắc chắn về overfitting chỉ từ các chỉ số hiện có.
