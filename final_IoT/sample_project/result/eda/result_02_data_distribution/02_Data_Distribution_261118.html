for folder in SEARCH_ROOTS:

        folder / "result" / "preprocessing" / "raw data" / "RT_IOT2022.csv",
        folder / "final_IoT" / "sample_project" / "result" / "preprocessing" / "raw data" / "RT_IOT2022.csv",

summary["Abs Skewness"] = summary["Skewness"].abs()

print("Kích thước dataset:", df.shape)

print(

print(

    (num.eq(0).mean() >= 0.95).sum()

display(
    summary.sort_values(
        "Abs Skewness",
        ascending=False
    ).head(15).round(3)
)

summary.to_csv(
    TABLE_DIR / "02_numeric_distribution_summary.csv",
    encoding="utf-8-sig"
)

selected = [
    "flow_duration",
    "flow_iat.tot",
    "bwd_pkts_tot",
    "flow_pkts_per_sec",
    "bwd_bulk_bytes",
    "active.std"
]

selected = [c for c in selected if c in num.columns]

fig, axes = plt.subplots(2, 3, figsize=(16, 9))

for ax, col in zip(axes.flat, selected):

    s = num[col].dropna()
    p99 = s.quantile(0.99)

    if p99 == 0:
        # Khi 99% dữ liệu bằng 0,
        # so sánh số mẫu bằng 0 và khác 0
        ax.bar(
            ["= 0", "!= 0"],
            [
                int((s == 0).sum()),
                int((s != 0).sum())
            ]
        )

        ax.set_title(f"{col} (P99 = 0)")

    else:
        # Chỉ giới hạn vùng hiển thị để dễ quan sát
        visible = s[s <= p99]

        ax.hist(
            visible.to_numpy(),
            bins=40
        )

        ax.set_title(f"{col} (hiển thị <= P99)")
        ax.set_xlabel(col)

    ax.set_ylabel("Số mẫu")

for ax in axes.flat[len(selected):]:
    ax.axis("off")

plt.suptitle(
    "EDA 02 - Histogram các đặc trưng",
    y=1.01
)

plt.tight_layout()

plt.savefig(
    FIG_DIR / "02_histogram_distributions.png",
    dpi=150,
    bbox_inches="tight"
)

plt.show()

3. So sánh phân phối gốc với Log1p¶

In [3]:
features_log_view = [
    "flow_iat.tot",
    "bwd_pkts_tot",
    "bwd_bulk_bytes"
]

features_log_view = [
    c for c in features_log_view if c in num.columns
]

fig, axes = plt.subplots(
    len(features_log_view),
    2,
    figsize=(14, 11)
)

if len(features_log_view) == 1:
    axes = np.array([axes])

for row, col in enumerate(features_log_view):

    s = num[col].dropna()

    # log1p chỉ áp dụng cho các giá trị không âm
    assert (s >= 0).all(), (
        f"Cột {col} chứa giá trị âm, không phù hợp log1p"
    )

    # Histogram gốc
    axes[row, 0].hist(
        s.to_numpy(),
        bins=50
    )

    axes[row, 0].set_title(
        f"{col} - Phân phối gốc"
    )
    axes[row, 0].set_ylabel("Số mẫu")

    # Histogram sau khi đổi thang hiển thị
    axes[row, 1].hist(
        np.log1p(s.to_numpy()),
        bins=50
    )

    axes[row, 1].set_title(
        f"{col} - Trực quan hóa Log1p"
    )
    axes[row, 1].set_xlabel(
        f"log(1 + {col})"
    )
    axes[row, 1].set_ylabel("Số mẫu")

plt.tight_layout()

plt.savefig(
    FIG_DIR / "02_original_vs_log1p_view.png",
    dpi=150,
    bbox_inches="tight"
)

plt.show()

Nhận xét: So sánh phân phối gốc và Log1p

Kết quả phân tích:

Ở thang đo gốc, các giá trị cực lớn kéo giãn trục ngang, khiến phần lớn mẫu tập trung gần giá trị 0 trên biểu đồ. Khi sử dụng thang Log1p, phạm vi giá trị được nén lại, giúp quan sát rõ hơn một số vùng phân bố của dữ liệu. Đối với các đặc trưng chứa nhiều số 0, Log1p không làm giảm tỷ lệ giá trị bằng 0.

Kết luận:

Các thuộc tính phân phối lệch phải có thể là ứng viên để thử nghiệm Log Transformation.

Tuy nhiên, không phải mọi đặc trưng đều phù hợp với phép biến đổi này.

from scipy.stats import gaussian_kde

col = "flow_iat.tot"

s = num[col].dropna()

# Chỉ lấy giá trị dương để khảo sát riêng
s_positive = s[s > 0]

# Lấy mẫu để vẽ nhanh, không thay đổi dữ liệu gốc
sample = s_positive.sample(
    n=min(20000, len(s_positive)),
    random_state=42
)

# Đổi thang hiển thị
view = np.log1p(sample)

fig, axes = plt.subplots(
    1, 2, figsize=(14, 4.5)
)

# Histogram
axes[0].hist(
    view.to_numpy(),
    bins=50,
    density=True
)

axes[0].set_title(
    "Histogram - log1p(flow_iat.tot)"
)
axes[0].set_xlabel("log1p(flow_iat.tot)")

# KDE
xs = np.linspace(
    view.min(),
    view.max(),
    400
)

kde = gaussian_kde(view.to_numpy())

axes[1].plot(
    xs,
    kde(xs)
)

axes[1].set_title(
    "KDE - log1p(flow_iat.tot)"
)
axes[1].set_xlabel("log1p(flow_iat.tot)")

plt.tight_layout()

plt.savefig(
    FIG_DIR / "02_histogram_kde_flow_iat.png",
    dpi=150,
    bbox_inches="tight"
)

plt.show()

print(
    f"Tỷ lệ {col} = 0:",
    f"{(s == 0).mean() * 100:.2f}%"
)

print(
    "Số mẫu dương dùng để vẽ:",
    len(sample)
)

Nhận xét: KDE Plot

Kết quả phân tích:

KDE giúp quan sát hình dạng phân phối và những vùng dữ liệu có mật độ cao. Biểu đồ được xây dựng từ các giá trị dương của flow_iat.tot sau khi chuyển thang Log1p. Các giá trị bằng 0 không được đưa vào KDE nên biểu đồ không đại diện cho toàn bộ phân phối ban đầu.

Kết luận:

Cần kết hợp KDE với Histogram, các phân vị và tỷ lệ giá trị bằng 0 để đánh giá chính xác hơn hình dạng phân phối.

cat_cols = [
