features = [
    "flow_duration",
    "flow_pkts_per_sec",
    "bwd_pkts_tot"
]

# Thứ tự lớp theo số lượng mẫu
class_order = df[target].value_counts().index.tolist()

# Trung vị và phân vị 90 của từng đặc trưng theo lớp
medians = (
    df.groupby(target)[features]
    .median()
    .add_suffix("_Median")
)

p90 = (
    df.groupby(target)[features]
    .quantile(0.90)
    .add_suffix("_P90")
)

class_statistics = pd.concat(
    [medians, p90],
    axis=1
).loc[class_order]

print("=== THỐNG KÊ ĐẶC TRƯNG THEO NHÃN ===")
display(class_statistics.round(4))

feature = "flow_duration"

# Tính IQR trên toàn dataset để khảo sát
q1 = df[feature].quantile(0.25)
q3 = df[feature].quantile(0.75)
iqr = q3 - q1

lower = q1 - 1.5 * iqr
upper = q3 + 1.5 * iqr

# Đánh dấu mẫu nằm ngoài ngưỡng IQR toàn cục
outlier_mask = (
    (df[feature] < lower) |
    (df[feature] > upper)
)

# Tổng hợp tỷ lệ ngoại lệ theo lớp
outliers_by_class = (
    df.assign(is_outlier=outlier_mask)
    .groupby(target)["is_outlier"]
    .agg(["sum", "count", "mean"])
)

outliers_by_class.columns = [
    "Outlier Count",
    "Total Samples",
    "Outlier Ratio"
]

outliers_by_class["Outlier (%)"] = (
    outliers_by_class["Outlier Ratio"] * 100
)

outliers_by_class = outliers_by_class.sort_values(
    "Outlier (%)",
    ascending=False
)

print("=== NGƯỠNG IQR ===")
print("Q1:", q1)
print("Q3:", q3)
print("IQR:", iqr)
print("Lower:", lower)
print("Upper:", upper)

display(outliers_by_class.round(2))

plt.figure(figsize=(12, 6))

Nhận xét: Ngoại lệ theo từng loại tấn công¶

Kết quả phân tích:

Khi sử dụng ngưỡng IQR toàn cục cho đặc trưng flow_duration:

  • Khoảng 99,8% mẫu MQTT_Publish nằm ngoài ngưỡng IQR.
  • Khoảng 99,4% mẫu DDOS_Slowloris nằm ngoài ngưỡng IQR.
  • Khoảng 97,8% mẫu Thing_Speak nằm ngoài ngưỡng IQR.
  • Trong khi đó, tỷ lệ tương ứng của DOS_SYN_Hping chỉ khoảng 0,1%.

Vấn đề phát hiện:

  1. Phân phối flow_duration khác biệt rất lớn giữa các loại lưu lượng mạng.
  2. Ngưỡng IQR toàn cục bị ảnh hưởng mạnh bởi đặc điểm của lớp đa số.
  3. Nhiều mẫu thuộc các lớp khác bị đánh dấu là ngoại lệ dù có thể là dữ liệu hợp lệ.
  4. Nếu loại bỏ toàn bộ mẫu ngoài ngưỡng IQR, nguy cơ mất phần lớn dữ liệu của một số lớp là rất cao.

port_features = [
    "id.orig_p",
    "id.resp_p"

records = []

for feature in port_features:

    for label, group in df.groupby(target):

        counts = group[feature].value_counts()

        most_common_port = counts.index[0]
        most_common_count = counts.iloc[0]

        records.append({
            "Feature": feature,
            "Attack_type": label,
            "Most Common Port": most_common_port,
            "Count": most_common_count,
            "Total": len(group),

Nhận xét: Mutual Information¶

Kết quả phân tích:

  • Các đặc trưng có điểm Mutual Information khác nhau, cho thấy mức độ liên hệ riêng lẻ với nhãn không giống nhau.
  • Những đặc trưng liên quan đến payload, thời gian và số gói tin xuất hiện trong nhóm có MI cao.
  • Một số đặc trưng có MI thấp, cần tiếp tục đánh giá mức độ hữu ích của chúng.

Vấn đề phát hiện:

  1. Không phải tất cả đặc trưng đều cung cấp lượng thông tin như nhau về nhãn.
  2. Một số đặc trưng có thể chứa thông tin tương tự nhau, dẫn đến các điểm MI cùng cao dù tương quan lớn.
  3. Mutual Information đánh giá từng đặc trưng riêng lẻ, không phản ánh đầy đủ khả năng kết hợp giữa nhiều đặc trưng.
  4. Điểm MI phụ thuộc vào tập mẫu được khảo sát. Các lớp ít mẫu có thể cho kết quả kém ổn định.

Kết luận:

Mutual Information giúp nhận diện những đặc trưng có mối liên hệ đáng chú ý với nhãn.

Tuy nhiên, chưa thể kết luận cần loại bỏ những đặc trưng có MI thấp.

Khi thực hiện Feature Selection thật sự, cần tính MI trên tập Train và đánh giá hiệu quả trên tập Validation.
